// =====================================================================
// 趋势跟踪信号（辅助指标）—— 文华财经，共 4 个，分别新建为独立指标。
// KAMA_MID 建为主图指标，其余 3 个建为副图指标。
// 每个指标从“===== 指标 N”下面一行复制到下一个分隔线之前。
// =====================================================================


// ===== 指标 1：KAMA_MID（主图，KAMA 与 MID = EMA(C,5)） =====
ERN:=10; FASTN:=2; SLOWN:=30; MIDN:=5;
DIR:=ABS(C-REF(C,ERN));
VIR:=SUM(ABS(C-REF(C,1)),ERN);
ER:=IFELSE(VIR>0,DIR/VIR,0);
FSC:=2/(FASTN+1);
SSC:=2/(SLOWN+1);
SC:=IFELSE(BARPOS>ERN,(ER*(FSC-SSC)+SSC)*(ER*(FSC-SSC)+SSC),0.99999);
KAMA:DMA(C,SC),COLORYELLOW;
MID:EMA(C,MIDN),COLORWHITE;


// ===== 指标 2：BB_Width_Angle（副图，>0 布林带张口，<0 收口） =====
BOLLN:=20; BOLLK:=2;
MB:=MA(C,BOLLN);
W:=2*BOLLK*STD(C,BOLLN)/MB;
BBWA:ATAN((W/REF(W,1)-1)*100)*180/3.1415926,COLORYELLOW;
STICKLINE(BBWA>0,0,BBWA,COLORRED,0);
STICKLINE(BBWA<0,0,BBWA,COLORGREEN,0);
ZERO:0,COLORWHITE;


// ===== 指标 3：ATR_Angle（副图，>0 波动放大，<0 波动收敛） =====
ATRN:=20;
TR1:=IFELSE(BARPOS>1,MAX(MAX(H-L,ABS(H-REF(C,1))),ABS(L-REF(C,1))),H-L);
ATR1:=MA(TR1,ATRN);
ATRA:ATAN(ATR1-REF(ATR1,1))*180/3.1415926,COLORYELLOW;
STICKLINE(ATRA>0,0,ATRA,COLORRED,0);
STICKLINE(ATRA<0,0,ATRA,COLORGREEN,0);
ZERO:0,COLORWHITE;


// ===== 指标 4：DIFF_MK（副图，MID - KAMA，动量） =====
ERN:=10; FASTN:=2; SLOWN:=30; MIDN:=5;
DIR:=ABS(C-REF(C,ERN));
VIR:=SUM(ABS(C-REF(C,1)),ERN);
ER:=IFELSE(VIR>0,DIR/VIR,0);
FSC:=2/(FASTN+1);
SSC:=2/(SLOWN+1);
SC:=IFELSE(BARPOS>ERN,(ER*(FSC-SSC)+SSC)*(ER*(FSC-SSC)+SSC),0.99999);
DIFFMK:EMA(C,MIDN)-DMA(C,SC),COLORYELLOW;
ZERO:0,COLORWHITE;
